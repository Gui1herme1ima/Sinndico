# Fatia 4.6 — Drawer de Detalhe Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Substituir a coluna "Ações" das 6 listas do Bloco A (Solicitações, Encomendas,
Visitantes, Comida, Reservas, Moradores) por um drawer lateral de detalhe, acionado pelo clique na
linha, fechando o Bloco A do roadmap (`docs/roadmap.md` linhas 51-52).

**Architecture:** Todas as mudanças são em `apps/web/src` (React + TS + Tailwind), sem tocar
backend — usa os dados/endpoints já existentes. 1 componente genérico novo (`Drawer`), 6
componentes de detalhe novos (um por módulo, presentational, sem fetch/mutation própria — recebem
dados e callbacks via props), edições pontuais nas 6 páginas de listagem pra ativar
`onRowClick`/`selectedRowKey` do `DataTable` (já existentes desde a 4.4.1, nunca ligados) e mover a
UI de ações pra dentro do drawer. `EditarMoradorModal.tsx` é removido (substituído pelo drawer).

**Tech Stack:** React 18, TypeScript, TanStack Query, Tailwind (tokens em `tokens.css`). Sem testes
automatizados nas páginas de listagem hoje (confirmado por busca prévia) — verificação por
`npm run typecheck` + teste manual no navegador, mesmo padrão da fatia 4.5.

## Global Constraints

- Nunca hardcodar cor hex fora de `apps/web/src/styles/tokens.css` — sempre classes Tailwind
  mapeadas pra tokens (`bg-surface`, `text-text-primary`, `bg-text-primary/40`, etc.).
- Transições seguem o padrão já usado no projeto:
  `transition-[...] duration-200 ease-[cubic-bezier(0.4,0,0.2,1)]`, respeitando
  `motion-reduce`/`motion-safe` onde aplicável (ver `Button.tsx`, `ThemeToggle.tsx`).
- `tsconfig.json` tem `noUnusedLocals: true` e `noUnusedParameters: true` — toda vez que uma UI de
  ação sai de uma página pro componente de detalhe, remover da página os imports que ficaram sem
  uso (`Button`, `Select`, etc.) e não deixar parâmetros de função não usados.
- Componentes de detalhe (`*Detail.tsx`) não fazem fetch nem `useMutation` própria — recebem os
  dados da linha e callbacks (`onChangeStatus`, `onSign`, etc.) via props; a página pai continua
  dona das mutations (mesmas já existentes, reaproveitadas).
- Nenhuma mudança de schema/API — usa os campos já expostos em
  `apps/web/src/services/api/types.ts`.
- Ao final de cada task: `npm run typecheck` (dentro de `apps/web`) tem que passar antes de
  commitar.
- Testar todo componente visual novo nos dois temas (claro/escuro) antes de considerar a task
  pronta — regra do `docs/escopo.md` seção 3.
- Estado da linha selecionada é guardado como **id** (`useState<string | null>`), nunca como o
  objeto da linha inteira — o objeto correspondente é derivado de `data.items.find(...)` a cada
  render, assim o drawer reflete o estado novo automaticamente depois que uma mutation invalida a
  query (requisito do spec: "drawer permanece aberto mostrando o novo estado").

```tsx
import { useEffect } from 'react';
import type { ReactNode } from 'react';

import { XIcon } from '@/components/ui/icons';
import { cn } from '@/lib/cn';

export interface DrawerProps {
  open: boolean;
  onClose: () => void;
  title: string;
  children: ReactNode;
}

export function Drawer({ open, onClose, title, children }: DrawerProps) {
  useEffect(() => {
    if (!open) return;
    function handleKeyDown(event: KeyboardEvent) {
      if (event.key === 'Escape') onClose();
    }
    document.addEventListener('keydown', handleKeyDown);
    return () => document.removeEventListener('keydown', handleKeyDown);
  }, [open, onClose]);

  return (
    <div
      className={cn('fixed inset-0 z-50 flex justify-end', open ? 'pointer-events-auto' : 'pointer-events-none')}
      aria-hidden={!open}
    >
      <div
        className={cn(
          'absolute inset-0 bg-text-primary/40 transition-opacity duration-200 ease-[cubic-bezier(0.4,0,0.2,1)]',
          open ? 'opacity-100' : 'opacity-0',
        )}
        onClick={onClose}
      />
      <div
        role="dialog"
        aria-modal="true"
        aria-label={title}
        className={cn(
          'relative flex h-full w-full flex-col bg-surface shadow-xl',
          'transition-transform duration-200 ease-[cubic-bezier(0.4,0,0.2,1)] motion-reduce:transition-none',
          'md:w-[420px]',
          open ? 'translate-x-0' : 'translate-x-full',
        )}
      >
        <div className="flex items-center justify-between border-b border-border p-4 md:p-6">
          <h3 className="font-display text-lg font-semibold text-text-primary">{title}</h3>
          <button
            type="button"
            onClick={onClose}
            aria-label="Fechar"
            className="text-text-secondary hover:text-text-primary"
          >
            <XIcon width={18} height={18} />
          </button>
        </div>
        <div className="flex-1 overflow-y-auto p-4 md:p-6">{children}</div>
      </div>
    </div>
  );
}
```

import { displayStatus } from '@/components/Visitantes/visitanteLabels';
import { Badge } from '@/components/ui/Badge';
import { Button } from '@/components/ui/Button';
import { formatResidencia } from '@/components/ui/MoradorSelect';
import { formatDate } from '@/lib/formatDate';
import type { MoradorDiretorioResponse, VisitanteResponse } from '@/services/api/types';

export interface VisitanteDetailProps {
  visitante: VisitanteResponse;
  morador: MoradorDiretorioResponse | undefined;
  canManage: boolean;
  pending: boolean;
  onEntrada: () => void;
  onSaida: () => void;
  onBloquear: () => void;
}

export function VisitanteDetail({
  visitante,
  morador,
  canManage,
  pending,
  onEntrada,
  onSaida,
  onBloquear,
}: VisitanteDetailProps) {
  const { status, label } = displayStatus(visitante);

  return (
    <div className="flex flex-col gap-4">
      <div>
        <h4 className="font-display text-lg font-semibold text-text-primary">{visitante.nomeVisitante}</h4>
        <p className="text-sm text-text-secondary">
          {[visitante.rg && `RG ${visitante.rg}`, visitante.placaVeiculo].filter(Boolean).join(' · ')}
        </p>
      </div>

      {morador && (
        <div className="text-sm text-text-secondary">
          <p className="font-medium text-text-primary">{morador.nome}</p>
          <p className="font-mono text-xs text-text-muted">{formatResidencia(morador.residencia)}</p>
        </div>
      )}

      <Badge status={status}>{label}</Badge>

      <div className="text-xs text-text-muted">
        <p>Visita {formatDate(visitante.dataVisita)}</p>
        {visitante.horaEntrada && <p>Entrada {formatDate(visitante.horaEntrada)}</p>}
        {visitante.horaSaida && <p>Saída {formatDate(visitante.horaSaida)}</p>}
      </div>

      {canManage && (visitante.status === 'aprovado' || (visitante.status === 'ativo' && !visitante.horaSaida)) && (
        <div className="flex flex-wrap gap-2 border-t border-border pt-4">
          {visitante.status === 'aprovado' && (
            <>
              <Button size="sm" loading={pending} onClick={onEntrada}>
                Registrar entrada

function labelResidencia(residencia: ResidenciaResponse): string {
  return residencia.bloco
    ? `Bloco ${residencia.bloco} — ${residencia.numero}`
    : `${residencia.rua}, ${residencia.numero}`;
}

export interface MoradorDetailPayload {
  nome: string;
  telefone?: string;
  residenciaId: string;
}

export interface MoradorDetailProps {
  morador: MoradorResponse;
  residencias: ResidenciaResponse[];

  hasError: boolean;
  onSave: (payload: MoradorDetailPayload) => void;
}

export function MoradorDetail({ morador, residencias, pending, hasError, onSave }: MoradorDetailProps) {
  const [nome, setNome] = useState(morador.nome);
  const [telefone, setTelefone] = useState(morador.telefone ?? '');
  const residenciaAtual = residencias.find(
    (r) =>
      r.bloco === morador.residencia?.bloco &&
      r.rua === morador.residencia?.rua &&
      r.numero === morador.residencia?.numero,
  );
  const [residenciaId, setResidenciaId] = useState(residenciaAtual?.id ?? residencias[0]?.id ?? '');

  return (
    <div className="flex flex-col gap-4">
      <div className="text-sm text-text-secondary">

      </div>

      <div className="flex flex-col gap-3 border-t border-border pt-4">
        <Input label="Nome" value={nome} onChange={(e) => setNome(e.target.value)} />
        <Input label="Telefone" value={telefone} onChange={(e) => setTelefone(e.target.value)} />
        <Select
          label="Residência"
          value={residenciaId}
          onChange={(e) => setResidenciaId(e.target.value)}
          options={residencias.map((r) => ({ value: r.id, label: labelResidencia(r) }))}
        />

        {hasError && <p className="text-sm text-danger">Não foi possível salvar. Tente de novo.</p>}

        <Button size="sm" loading={pending} onClick={() => onSave({ nome, telefone: telefone || undefined, residenciaId })}>
          Salvar
        </Button>
      </div>
    </div>
  );
}
